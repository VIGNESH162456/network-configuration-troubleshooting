! SW1 VLAN and trunk reference configuration
! Adjust interface names for your selected switch.
enable
configure terminal
hostname SW1

vlan 10
 name DEPARTMENT_A
 exit
vlan 20
 name DEPARTMENT_B
 exit

interface fastEthernet0/1
 description PC-A - VLAN 10
 switchport mode access
 switchport access vlan 10
 spanning-tree portfast
 exit

interface fastEthernet0/2
 description PC-B - VLAN 20
 switchport mode access
 switchport access vlan 20
 spanning-tree portfast
 exit

interface fastEthernet0/3
 description DNS server - VLAN 10
 switchport mode access
 switchport access vlan 10
 spanning-tree portfast
 exit

interface gigabitEthernet0/1
 description 802.1Q trunk to R1
 switchport mode trunk
 switchport trunk allowed vlan 10,20
 no shutdown
 exit

end
write memory
